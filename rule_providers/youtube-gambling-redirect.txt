# YouTube Sponsored App Gambling Redirect Guard v4.9
# Blocks suspicious ad destinations that commonly redirect from YouTube ads to gambling apps
# Does not block Google Play Store globally.

# Regex applies to redirect destinations only; never block YouTube playback hosts.
DOMAIN-REGEX,(?i)(^|\.)[^.]*((casino|casinogame|slot|slots|togel|judi|gacor|poker|jackpot|sbobet)[^.]*)(\.|$),REJECT
DOMAIN-REGEX,(?i)(^|\.)[^.]*((bet|betting|taruhan|deposit|withdraw|spin|bonus)[^.]*)(\.|$),REJECT
DOMAIN-REGEX,(?i)(^|\.)[^.]+\.(bet|casino)(\.|$),REJECT
